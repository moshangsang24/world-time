## 4. 核心算法实现

### 4.1 太阳位置计算算法
```typescript
/**
 * 计算指定时间的太阳位置
 * 基于天文算法，考虑地球椭圆轨道和倾斜角
 */
export function calculateSunPosition(dateTime: DateTime): SunPosition {
  const jd = getJulianDay(dateTime);
  const centuries = (jd - 2451545.0) / 36525.0;
  
  // 计算太阳几何平均经度
  const geomMeanLongSun = (280.46646 + centuries * (36000.76983 + centuries * 0.0003032)) % 360;
  
  // 计算太阳几何平均异常点
  const geomMeanAnomSun = 357.52911 + centuries * (35999.05029 - 0.0001537 * centuries);
  
  // 计算地球轨道偏心率
  const eccentEarth = 0.016708634 - centuries * (0.000042037 + 0.0000001267 * centuries);
  
  // 太阳中心方程
  const sunEq = Math.sin(toRadians(geomMeanAnomSun)) * (1.914602 - centuries * (0.004817 + 0.000014 * centuries)) +
                Math.sin(toRadians(2 * geomMeanAnomSun)) * (0.019993 - 0.000101 * centuries) +
                Math.sin(toRadians(3 * geomMeanAnomSun)) * 0.000289;
  
  // 太阳真实经度
  const sunTrueLong = geomMeanLongSun + sunEq;
  
  // 计算太阳赤纬角
  const obliqCorr = 23 + (26 + ((21.448 - centuries * (46.815 + centuries * (0.00059 - centuries * 0.001813)))) / 60) / 60;
  const sunDeclination = toDegrees(Math.asin(Math.sin(toRadians(obliqCorr)) * Math.sin(toRadians(sunTrueLong))));
  
  return {
    declination: sunDeclination,
    longitude: (sunTrueLong + 180) % 360 - 180,
    // ... 其他计算属性
  };
}
```

### 4.2 昼夜分界线计算
```typescript
/**
 * 计算昼夜分界线（晨昏线）
 */
export function calculateDayNightBoundary(sunPosition: SunPosition): DayNightBoundary {
  const points: Array<{lat: number, lng: number}> = [];
  const { declination, longitude } = sunPosition;
  
  // 计算每个经度上的昼夜分界点
  for (let lng = -180; lng <= 180; lng += 1) {
    const hourAngle = lng - longitude;
    
    // 计算该经度上日出日落时的纬度
    const cosHourAngle = -Math.tan(toRadians(declination)) * Math.tan(toRadians(0)); // 地平线
    
    if (Math.abs(cosHourAngle) <= 1) {
      const lat = toDegrees(Math.atan(-cosHourAngle / Math.sin(toRadians(hourAngle))));
      points.push({ lat, lng });
    }
  }
  
  return {
    points,
    sunLongitude: longitude,
    isDaylight: (lat: number, lng: number) => {
      // 判断指定坐标是否为白昼
      const hourAngle = lng - longitude;
      const sunAltitude = calculateSunAltitude(lat, declination, hourAngle);
      return sunAltitude > -0.833; // 考虑大气折射
    }
  };
}
```

### 4.3 时间滚动控制
```typescript
/**
 * 滚动事件处理，实现时间控制
 */
export function useScrollTimeControl() {
  const { timeOffset, setTimeOffset, isRealTime, toggleRealTime } = useTimeStore();
  
  const handleWheel = useCallback((event: WheelEvent) => {
    event.preventDefault();
    
    if (isRealTime) {
      toggleRealTime(); // 退出实时模式
    }
    
    const delta = event.deltaY;
    const sensitivity = event.ctrlKey ? 0.1 : 1; // Ctrl键精细控制
    const newOffset = timeOffset + (delta > 0 ? sensitivity : -sensitivity);
    
    // 限制在±24小时范围内
    const clampedOffset = Math.max(-1440, Math.min(1440, newOffset));
    setTimeOffset(clampedOffset);
  }, [timeOffset, isRealTime, setTimeOffset, toggleRealTime]);
  
  useEffect(() => {
    window.addEventListener('wheel', handleWheel, { passive: false });
    return () => window.removeEventListener('wheel', handleWheel);
  }, [handleWheel]);
}
```

## 5. 性能优化策略

### 5.1 渲染性能优化
- **LOD (Level of Detail)**: 根据视距调整地球模型精度
- **实例化渲染**: 城市标记使用实例化渲染
- **纹理压缩**: 使用WebP格式压缩纹理
- **视锥体剔除**: 只渲染可见区域的对象

### 5.2 计算性能优化
- **时间缓存**: 缓存重复的时间计算结果
- **太阳位置缓存**: 按分钟级别缓存太阳位置
- **Web Workers**: 复杂天文计算移至后台线程
- **防抖节流**: 滚动事件使用节流控制

### 5.3 内存管理
- **纹理池**: 复用纹理对象
- **几何体缓存**: 缓存常用几何体
- **组件卸载**: 及时清理Three.js资源
- **弱引用**: 使用WeakMap存储临时数据

## 9. 风险评估与应对

### 9.1 技术风险
- **3D渲染性能**: 在低端设备上可能卡顿
  - 应对: 提供2D备选模式，自适应质量调整
- **天文计算精度**: 计算复杂度高
  - 应对: 使用成熟库，适当的精度权衡
- **浏览器兼容性**: WebGL支持问题
  - 应对: 兼容性检测，降级方案

### 9.2 用户体验风险
- **学习成本**: 滚动控制可能不直观
  - 应对: 添加引导动画和说明文档
- **加载时间**: 3D资源较大
  - 应对: 懒加载，渐进式加载
